# Park Day family-sync API — paste into the j5rescue.com vhost in
# CloudPanel (Sites → j5rescue.com → Vhost Editor), inside the main
# `server { }` block, ABOVE the existing `location /` and PHP blocks.

location ^~ /FamilyFunWeek/api/ {
    proxy_pass http://127.0.0.1:8787/;
    proxy_http_version 1.1;
    proxy_set_header Host $host;
    proxy_set_header X-Real-IP $remote_addr;
    proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    proxy_set_header X-Forwarded-Proto $scheme;
    proxy_read_timeout 15s;
    client_max_body_size 3m;
}
